// Получаем ID для редактирования const editId = parseInt(localStorage.getItem('editId')); // Загружаем товар при загрузке страницы window.addEventListener('DOMContentLoaded', function() { if (!editId) { alert('Товар не выбран'); window.location.href = 'index.html'; return; } // Загружаем товары const products = JSON.parse(localStorage.getItem('products')) || []; // Ищем товар const product = products.find(p => p.id === editId); if (!product) { alert('Товар не найден. ID: ' + editId); window.location.href = 'index.html'; return; } // Заполняем форму document.getElementById('productId').value = product.id; document.getElementById('name').value = product.name || ''; document.getElementById('price').value = product.price || ''; document.getElementById('description').value = product.description || ''; document.getElementById('image').value = product.image || ''; // Превью изображения const imagePreview = document.getElementById('imagePreview'); if (product.image && product.image !== 'https://via.placeholder.com/200x150?text=No+Image') { imagePreview.src = product.image; imagePreview.style.display = 'block'; } }); // Обработка отправки формы document.getElementById('editProductForm').addEventListener('submit', function(e) { e.preventDefault(); const productId = parseInt(document.getElementById('productId').value); let products = JSON.parse(localStorage.getItem('products')) || []; // Находим индекс const index = products.findIndex(p => p.id === productId); if (index === -1) { alert('Товар не найден'); return; } // Обновляем товар products[index] = { id: productId, name: document.getElementById('name').value.trim(), price: parseFloat(document.getElementById('price').value), description: document.getElementById('description').value.trim(), image: document.getElementById('image').value.trim() || 'https://via.placeholder.com/200x150?text=No+Image' }; // Сохраняем localStorage.setItem('products', JSON.stringify(products)); alert('Товар успешно обновлен!'); window.location.href = 'index.html'; }); // Удаление товара function deleteProduct() { if (!confirm('Удалить этот товар?')) { return; } const productId = parseInt(document.getElementById('productId').value); let products = JSON.parse(localStorage.getItem('products')) || []; products = products.filter(p => p.id !== productId); localStorage.setItem('products', JSON.stringify(products)); alert('Товар удален!'); window.location.href = 'index.html'; } // Превью изображения document.getElementById('image').addEventListener('input', function(e) { const imagePreview = document.getElementById('imagePreview'); if (e.target.value.trim()) { imagePreview.src = e.target.value; imagePreview.style.display = 'block'; } });
